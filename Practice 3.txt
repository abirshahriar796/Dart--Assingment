2. Print even numbers between intervals using function
void printEvenNumbers(int start, int end) {
  for (int i = start; i <= end; i++) {
    if (i % 2 == 0) {
      print(i);
    }
  }
}

void main() {
  printEvenNumbers(1, 20);
}

8
10

4. Generate random password
import 'dart:math';

String generatePassword(int length) {
  const characters =
      'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789';

  Random random = Random();

  String password = '';

  for (int i = 0; i < length; i++) {
    password += characters[random.nextInt(characters.length)];
  }

  return password;
}

void main() {
  String password = generatePassword(8);

  print("Random Password: $password");
}

Sample Output:

Random Password: A7kP2mQ9

6. Reverse a String using function

}

}

9. Function maxNumber to find largest of three numbers
int maxNumber(int a, int b, int c) {
  int max = a;

  if (b > max) {
    max = b;
  }

  if (c > max) {
    max = c;
  }

  return max;
}

void main() {
  int result = maxNumber(10, 25, 15);

  print("Largest Number = $result");
}

Output:

Largest Number = 25
